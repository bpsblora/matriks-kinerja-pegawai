function distributionForDates(start,end){

 const counts=Array(13).fill(0),calendar=Array(13).fill(0);
 for(let t=first.getTime();t<=last.getTime();t+=86400000){const d=new Date(t),month=d.getUTCMonth()+1;calendar[month]++;if(workingDate(d.toISOString().slice(0,10)))counts[month]++}
 const work=counts.reduce((a,b)=>a+b,0),weights=work?counts:calendar,total=weights.reduce((a,b)=>a+b,0);
 if(!total)return null;
 return {months:weights.map((x,m)=>({m,days:x,share:x/total})).filter(x=>x.m>0&&x.days>0),workDays:work,days:calendar.reduce((a,b)=>a+b,0),fallback:work===0};
}

function groupPersonWorkload(rows){
 const groups=new Map();
 for(const row of rows){
  const ac=row.ac,h=Number(row.h)||0;
  if(!ac||!h)continue;
  const team=String(ac.team||teamMap.get(ac.team_id)?.name||'Tim Belum Ditetapkan');
  const activity=groupName(ac);
  const identifier=String(ac.team_id||team)+'|'+activity;
  let group=groups.get(identifier);
  if(!group){
   group={key:identifier,team,activity,hours:0,items:[]};
   groups.set(identifier,group);
  }
  group.hours+=h;
  group.items.push({id:ac.id||'',title:String(ac.description||'Uraian Tanpa Nama'),hours:h});
 }
 return [...groups.values()].map(g=>({
  ...g,
  items:g.items.sort((a,b)=>b.hours-a.hours||a.title.localeCompare(b.title,'id'))
 })).sort((a,b)=>b.hours-a.hours||a.team.localeCompare(b.team,'id')||a.activity.localeCompare(b.activity,'id'));
}
