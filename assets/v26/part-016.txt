function kroRecommendation(k,tid){
 // Recommendations use work/activity and KRO *names*, not a permanent KRO code.
 // Always require an explicit annual assignment/confirmation.
 let topGroup=null,bestGroup=0;
 for(const [group] of groupList(tid,true)){
  const score=kroSimilarity(group,k.name);
  if(score>bestGroup){topGroup=group;bestGroup=score}
 }
 if(bestGroup>=.67&&topGroup)return {
  kind:'serupa',label:'Nama Kegiatan Sesuai',
  reason:'Nama Kegiatan "'+topGroup+'" mirip dengan Nama KRO "'+k.name+'". Periksa cakupan POK '+activeYear+' sebelum menetapkan.'
 };
 const previous=previousYearKroReference(tid);
 let closest=null,best=0;
 for(const earlier of previous){
  const score=kroSimilarity(earlier.name,k.name);
  if(score>best){closest=earlier;best=score}
 }
 if(best>=.67&&closest)return {
  kind:'serupa',label:'Nama KRO Tahun Lalu Serupa',
  reason:'Nama KRO '+(activeYear-1)+' "'+closest.name+'" mirip dengan "'+k.name+'". Kode dapat berbeda; penetapan tahun '+activeYear+' tetap perlu konfirmasi.'
 };
 if((k.mappings||[]).some(m=>m.team_id===tid))return {
  kind:'pendukung',label:'Acuan Tim Historis',
  reason:'Terdapat referensi hubungan tim dan KRO pada data sumber. Ini bukan penetapan otomatis untuk '+activeYear+'.'
 };
 if(activeYear===2027){
  const support=REKOMENDASI_KRO_LINTAS_TIM[tid]?.[k.code];
  if(support)return {kind:'pendukung',label:'Usulan Tugas Tim',reason:support+' Periksa kesesuaian tugas dengan KRO tahun berjalan.'}
 }
 return null;
}

    snapshot=JSON.parse(localStorage.getItem(key)||'null');

  const ownedIds=new Set(rows.filter(a=>String(a.team_id)===teamId).map(a=>String(a.id)));
  // Use IDs rather than names: team and activity names may change each year.
  if([...Object.keys(snapshot.allocEdits||{}),...Object.keys(snapshot.workflowPlans||{})]
   .some(key=>ownedIds.has(String(key).split('|')[0])))return true;
 }

}

/* Activity ownership determines which team's workload receives the hours.
   Assigned employees may belong to any team. Composite activity keys ensure
   identical activity names under different teams are never conflated. */
function trendAvailableActivities(){
 const byKey=new Map();
 for(const item of activities){
  const tid=String(item.team_id),name=String(groupName(item)||'').trim();
  if(!name||(trendOwnerTeams instanceof Set&&!trendOwnerTeams.has(tid)))continue;
  const key=trendActivityId(tid,name);
  if(!byKey.has(key))byKey.set(key,{key,tid,name,team:teamMap.get(tid)?.name||BASE.teams.find(t=>String(t.id)===tid)?.name||tid});
 }
 return [...byKey.values()].sort((a,b)=>a.team.localeCompare(b.team,'id')||a.name.localeCompare(b.name,'id'));
}
function trendWorkloadMonth(month,ownerTeams=null,activityKeys=null){
 const result=periodAgg(String(month),''),totals=new Map(),teamTotals=new Map();
 for(const [key,hours] of result.details){
  const split=key.lastIndexOf('|'),activityId=key.slice(0,split),employeeId=key.slice(split+1);
  const activity=byId.get(activityId)||baseActivity.get(activityId);
  if(!activity)continue;
  const teamId=String(activity.team_id),activityName=groupName(activity);
  if(ownerTeams instanceof Set?!ownerTeams.has(teamId):ownerTeams&&teamId!==String(ownerTeams))continue;
  if(activityKeys instanceof Set?!activityKeys.has(trendActivityId(teamId,activityName)):activityKeys&&activityName!==activityKeys)continue;
  totals.set(employeeId,(totals.get(employeeId)||0)+hours);
  teamTotals.set(teamId,(teamTotals.get(teamId)||0)+hours);
 }
 return {totals,teamTotals};
}

 if(!match)return '';
 const day=Number(match[1]),month=Number(match[2]),year=match[3].length===2?2000+Number(match[3]):Number(match[3]);

}

function kroAuditSnapshot(){
 const teamLinks=new Map();for(const k of POK_KROS)teamLinks.set(k.code,[]);
 for(const t of demoTeams)if(yearKroReady(t.id))for(const code of new Set(t.kroCodes||[]))if(teamLinks.has(code))teamLinks.get(code).push(t);
 const unassigned=POK_KROS.filter(k=>!teamLinks.get(k.code)?.length);
 const teamsWithoutKro=demoTeams.filter(t=>!yearKroReady(t.id)||!teamKroCodes(t.id).length);
 const visibleTeamsWithoutKro=currentRole==='ketua'?teamsWithoutKro.filter(t=>t.id===roleTeamId):teamsWithoutKro;
 const emptyPairs=[];
 const unmapped=[];
 for(const team of demoTeams){
  const groups=groupList(team.id,false);
  const counts=new Map();for(const code of teamKroCodes(team.id))counts.set(code,0);
  for(const [group,arr] of groups){
   const code=resolveGroupKro(team.id,group);
   if(code===NO_KRO||!counts.has(code))unmapped.push({tid:team.id,team:team.name,group,descriptions:arr.length});
   else counts.set(code,counts.get(code)+1);
  }
  for(const [code,count] of counts)if(count===0)emptyPairs.push({tid:team.id,team:team.name,code,k:POK_KROS.find(x=>x.code===code)});
 }
 const visiblePairs=currentRole==='ketua'?emptyPairs.filter(p=>p.tid===roleTeamId):emptyPairs;
 const visibleUnmapped=currentRole==='ketua'?unmapped.filter(p=>p.tid===roleTeamId):unmapped;
 return {unassigned,teamsWithoutKro,visibleTeamsWithoutKro,emptyPairs,unmapped,visiblePairs,visibleUnmapped,teamLinks};
}

function capacityWarningEvents(period){
 const periodName=period==='monthly'?'Tahun '+activeYear:months[Number(period)-1]+' '+activeYear;
 const permitted=isBaseYear?null:new Set(yearSetup.staff?.ids||[]);
 const staff=BASE.employees.filter(e=>e.type==='pegawai'&&e.active!==false&&(!permitted||permitted.has(e.id)));
 const perPerson=capacity(period),overall=periodAgg(period,'').totals;
 const events=[],mk=(level,scope,title,description,extra={})=>events.push({
  level,scope,title,description,period,periodName,...extra
 });
 const getHours=e=>overall.get(e.id)||0;
 const ratioText=(hours,cap)=>cap?nf(hours/cap*100)+'%':'—';
 const excess=(hours,cap)=>nf(Math.max(0,hours-cap))+' Jam';
 const remaining=(hours,cap)=>nf(Math.max(0,cap-hours))+' Jam';
 const teamLabel=name=>/^tim\b/i.test(String(name))?String(name):'Tim '+String(name);
 const officeHours=staff.reduce((sum,e)=>sum+getHours(e),0);
 const noAllocation=staff.filter(e=>getHours(e)<=0);
 const effectiveCalendarReady=typeof HOLIDAYS==='undefined'||!!HOLIDAYS[activeYear];
 if(!effectiveCalendarReady&&period==='monthly')mk('pending','alokasi','Kalender Hari Kerja '+activeYear+' Belum Diverifikasi',
  'Daftar libur nasional/cuti bersama tahun '+activeYear+' belum tersedia. Penilaian melebihi, mendekati, atau sangat rendah dibanding kapasitas efektif ditangguhkan agar tidak menyesatkan.',{action:'check-staff'});
 // A person has one effective work capacity across all assignments.
 // A team owns a workload; it does not have a fixed capacity denominator.
 // Under-allocation is provisional: 0 hours means absent records, not proven idle time.
 if(effectiveCalendarReady&&perPerson>0&&officeHours>0){
  for(const e of staff){
   const hours=getHours(e),pct=hours/perPerson;
   if(pct>1.00001)mk('over','pegawai',e.full_name+' — Melebihi Kapasitas Total Efektif '+excess(hours,perPerson),
    nf(hours)+' Jam Alokasi dari '+nf(perPerson)+' Jam Kapasitas Total Efektif ('+ratioText(hours,perPerson)+'). Evaluasi distribusi pekerjaan, jadwal, dan potensi kebutuhan lembur.',{employeeId:e.id,loadRatio:pct});
   else if(pct>=.85)mk('near','pegawai',e.full_name+' — Tersisa '+remaining(hours,perPerson)+' Kapasitas Efektif',
    nf(hours)+' Jam Alokasi dari '+nf(perPerson)+' Jam Kapasitas Total Efektif ('+ratioText(hours,perPerson)+').',{employeeId:e.id,loadRatio:pct});
   else if(hours>0&&pct<=.25)mk('under','pegawai',e.full_name+' — '+nf(hours)+' Jam Alokasi ('+nf(pct*100)+'% dari Total Efektif)',
    'Alokasi tercatat sangat rendah dibanding '+nf(perPerson)+' Jam Kapasitas Total Efektif. Periksa kelengkapan pencatatan sebelum menyimpulkan pegawai kurang pekerjaan.',{employeeId:e.id,loadRatio:pct});
  }
 }
 if(officeHours>0&&noAllocation.length){
  mk('pending','alokasi',nf(noAllocation.length)+' Pegawai Belum Memiliki Alokasi Tercatat',
   'Pada '+periodName+', terdapat '+nf(noAllocation.length)+' Pegawai dengan 0 Jam alokasi. Periksa kelengkapan pencatatan; ini bukan bukti pegawai tidak bekerja.',{action:'check-staff'});
 }
 // Monthly distribution: compare only teams with actual recorded work.
 // A zero-work team is not deemed idle without a known planned assignment.
 if(period!=='monthly'){
  const teams=BASE.teams.map(t=>{
   const aggregate=periodAgg(period,t.id),groups=new Map(),uniqueTasks=new Set();
   for(const [key,hours] of aggregate.details||new Map()){
    const id=key.split('|')[0],activity=byId.get(id);
    if(!activity)continue;
    uniqueTasks.add(id);
    const name=String(activity.group||'Kegiatan Lainnya');
    groups.set(name,(groups.get(name)||0)+(Number(hours)||0));
   }
   const mainActivities=[...groups.entries()].sort((a,b)=>b[1]-a[1]).slice(0,2)
    .map(([name,hours])=>name+' ('+nf(hours)+' Jam)');
   return {
    team:t,
    hours:[...aggregate.totals.values()].reduce((sum,h)=>sum+h,0),
    activityCount:groups.size,taskCount:uniqueTasks.size,mainActivities
   };
  });
  const working=teams.filter(row=>row.hours>0);
  if(working.length>=3){
   const mean=working.reduce((sum,row)=>sum+row.hours,0)/working.length;
   for(const row of working){
    const gap=row.hours-mean,ratio=mean?row.hours/mean:1;
    if(gap>=8&&ratio>=1.75){
     mk('pending','tim',teamLabel(row.team.name)+' — '+nf(gap)+' Jam di Atas Rata-rata Tim',
      nf(row.hours)+' Jam Beban Kegiatan dibanding rata-rata '+nf(mean)+' Jam dari '+working.length+' Tim yang memiliki alokasi pada '+periodName+'. Tercatat '+nf(row.activityCount)+' Kegiatan dan '+nf(row.taskCount)+' Uraian.'+(row.mainActivities.length?' Penyumbang terbesar: '+row.mainActivities.join(', ')+'.':' Periksa rincian kegiatan.') ,{teamId:row.team.id,loadRatio:ratio,anomaly:'high'});
    }else if(-gap>=8&&ratio<=.35){
     mk('pending','tim',teamLabel(row.team.name)+' — '+nf(-gap)+' Jam di Bawah Rata-rata Tim',
      nf(row.hours)+' Jam Beban Kegiatan dibanding rata-rata '+nf(mean)+' Jam dari '+working.length+' Tim yang memiliki alokasi pada '+periodName+'. Baru tercatat '+nf(row.activityCount)+' Kegiatan dan '+nf(row.taskCount)+' Uraian; periksa apakah kegiatan memang sedikit atau belum teralokasikan.',{teamId:row.team.id,loadRatio:ratio,anomaly:'low'});
    }
   }
  }
 }
 if(isBaseYear){
  if(period==='monthly'){
   const annual=periodAgg('annual','').totals;
   const reference=BASE.employees.filter(e=>e.type==='pegawai').reduce((n,e)=>n+(annual.get(e.id)||0),0);
   const current=BASE.employees.filter(e=>e.type==='pegawai').reduce((n,e)=>n+(overall.get(e.id)||0),0);
   if(Math.abs(reference-current)>=.005)mk('pending','referensi','Alokasi '+activeYear+' — '+nf(Math.abs(reference-current))+' Jam '+(reference>current?'Belum Terurai':'Melebihi Acuan'),
    'Selisih '+nf(Math.abs(reference-current))+' Jam ('+nf(reference?Math.abs(reference-current)/reference*100:100)+'%). Rincian bulanan dan acuan tahunan perlu diperiksa di Rekonsiliasi.',{action:'recon'});
  }
 }else{
  // Only assess work with a recorded monthly allocation, a positive target,
  // or a workflow plan. Catalogue activities alone are not missing allocations.
  const monthMatches=m=>period==='monthly'||Number(m)===Number(period);
  const assigned=new Map();
  const saved=new Map(baseMonthly);
  for(const [key,val] of Object.entries(state.allocEdits||{}))saved.set(key,Number(val)||0);
  const missingDate=new Map();
  for(const [allocKey,qty] of saved){
   if(!(Number(qty)>0))continue;
   const [aid,eid,rawMonth]=allocKey.split('|'),m=Number(rawMonth);
   if(!monthMatches(m))continue;
   const a=byId.get(aid);if(!a||a.status!=='aktif')continue;
   const lookup=aid+'|'+m;
   assigned.set(lookup,(assigned.get(lookup)||0)+1);
   const schedule=state.scheduleEdits?.[amKey(aid,m)];
   const dateOK=schedule&&/^\d{4}-\d{2}-\d{2}$/.test(schedule.start||'')
    &&/^\d{4}-\d{2}-\d{2}$/.test(schedule.end||'')
    &&schedule.start<=schedule.end
    &&schedule.start.slice(0,4)===String(activeYear)
    &&schedule.end.slice(0,4)===String(activeYear);
   if(!dateOK&&!missingDate.has(aid))missingDate.set(aid,{a,m});
  }
  const expected=new Map();
  for(const [key,target] of Object.entries(state.targets||{})){
   if(!(Number(target)>0))continue;
   const [aid,rawMonth]=key.split('|'),m=Number(rawMonth);
   if(monthMatches(m)&&byId.get(aid)?.status==='aktif')expected.set(aid+'|'+m,{a:byId.get(aid),m});
  }
  for(const [key,plan] of Object.entries(state.workflowPlans||{})){
   if(!plan||plan.status==='cancelled')continue;
   const [aid,rawMonth]=key.split('|'),m=Number(rawMonth);
   if(monthMatches(m)&&byId.get(aid)?.status==='aktif')expected.set(aid+'|'+m,{a:byId.get(aid),m});
  }
  const unassigned=[...expected.entries()].filter(([k])=>!assigned.has(k)).map(([,item])=>item);
  if(missingDate.size){
   const first=missingDate.values().next().value;
   mk('pending','alokasi',nf(missingDate.size)+' Uraian Belum Memiliki Tanggal Pelaksanaan',
    missingDate.size+' Uraian memiliki alokasi pegawai, tetapi tanggal mulai atau selesai belum lengkap.',{activityId:first.a.id,month:first.m,action:'input'});
  }
  if(unassigned.length){
   const first=unassigned[0];
   mk('pending','alokasi',nf(new Set(unassigned.map(x=>x.a.id)).size)+' Uraian Belum Dialokasikan ke Pegawai',
    new Set(unassigned.map(x=>x.a.id)).size+' Uraian yang sudah direncanakan belum memiliki alokasi pegawai.',{activityId:first.a.id,month:first.m,action:'input'});
  }
 }
 return {events,hasData:officeHours>0||events.length>0};
}

function warningCollection(){
 const periods=['monthly',...months.map((_,i)=>String(i+1))];
 const snapshots=periods.map(period=>capacityWarningEvents(period));
 const all=snapshots.flatMap(s=>s.events);
 const priority={over:0,pending:1,near:2,under:3};
 const order=(a,b)=>{
  const importance=priority[a.level]-priority[b.level];
  if(importance)return importance;
  if(a.level==='over'||a.level==='near'){
   const severity=(b.loadRatio||0)-(a.loadRatio||0);
   if(Math.abs(severity)>0.00001)return severity;
  }
  const number=p=>p==='monthly'?0:Number(p);
  return number(a.period)-number(b.period)||a.title.localeCompare(b.title,'id')||a.description.localeCompare(b.description,'id');
 };
 all.sort(order);
 const totals={over:0,near:0,pending:0,under:0};
 all.forEach(e=>totals[e.level]++);
 return {all,totals,hasData:snapshots.some(s=>s.hasData)};
}

  if(existing){existing.code=code;existing.name=name;existing.last_edited_at=new Date().toISOString();existing.master_edited=true}
  else POK_KROS.push({code,name,mappings:[],source_row:null,details:[],mapping_status:'Ditambahkan manual',master_created:true});
  POK_KROS.sort((a,b)=>a.code.localeCompare(b.code,'id'));
  if(!saveKroMaster())return err.textContent='Gagal menyimpan ke penyimpanan lokal browser.';
  logChange(oldCode?'Edit daftar induk KRO':'Tambah KRO '+activeYear,(oldCode||code)+' → '+code+' · '+name);
  persist();d.close();
  const panel=$('kroBoardMain');if(panel)panel.dataset.team='';
  renderTeamAdminPage();renderKroBoard();renderKroAudit();
  toast(oldCode?'KRO diperbarui.':'KRO berhasil ditambahkan ke daftar induk '+activeYear+'.');
 };
 return d;
}

function kroCoverageSnapshot(previewTeamId='',previewCodes=null){
 const validCodes=new Set(POK_KROS.map(k=>k.code));
 const assigned=new Map(),covered=new Set();
 for(const team of demoTeams){
  const proposed=previewCodes!==null&&team.id===previewTeamId;
  const codes=(proposed?[...previewCodes]:(yearKroReady(team.id)?teamKroCodes(team.id):[])).filter(code=>validCodes.has(code));
  assigned.set(team.id,codes);for(const code of codes)covered.add(code);
 }
 const teamsMissing=demoTeams.filter(team=>!(assigned.get(team.id)||[]).length);
 const krosMissing=POK_KROS.filter(k=>!covered.has(k.code));
 return {teamsMissing,krosMissing,assignedTeamCount:demoTeams.length-teamsMissing.length,assignedKroCount:POK_KROS.length-krosMissing.length,teamCount:demoTeams.length,kroCount:POK_KROS.length};
}
