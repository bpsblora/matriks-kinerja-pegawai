function kroRecommendation(k,tid){
 if(activeYear>=2028){
  try{const old=JSON.parse(localStorage.getItem('bps-blora-v19-team-store-'+(activeYear-1)));if(old?.find(t=>t.id===tid)?.kroCodes?.includes(k.code))return {kind:'langsung',label:'Acuan tahun sebelumnya',reason:'KRO sama dengan penugasan tim '+(activeYear-1)};}catch(e){}
 }
 if((k.mappings||[]).some(m=>m.team_id===tid))return {kind:'langsung',label:'Acuan 2026',reason:'Ada kesesuaian pemetaan tim dari matriks 2026 dan referensi POK 2027.'};
 if(activeYear>=2027&&activeYear<=2027){
  const old=KRO_2026_REFERENCE.filter(k=>k.mappings?.some(m=>m.team_id===tid));
  const exact=old.find(p=>p.code===k.code);
  if(exact)return {kind:'langsung',label:'Sama seperti 2026',reason:'Kode KRO tercantum untuk tim ini pada referensi 2026.'};
  let closest=null,best=0;
  for(const p of old){const v=kroSimilarity(p.name,k.name);if(v>best){closest=p;best=v}}
  if(best>=0.66&&closest)return {kind:'serupa',label:'KRO serupa 2026',reason:'Mirip dengan '+closest.code+' — '+closest.name+' (2026); mohon periksa cakupan pekerjaan.'};
  const support=REKOMENDASI_KRO_LINTAS_TIM[tid]?.[k.code];
  if(support)return {kind:'pendukung',label:'Usulan dukungan lintas tim',reason:support+' Bukan KRO khusus milik tim; perlu konfirmasi.'};
 }
 return null;
}

function displayDate(iso){if(!/^\d{4}-\d{2}-\d{2}$/.test(iso||''))return '';const [y,m,d]=iso.split('-');return d+'/'+m+'/'+y.slice(2)}
function parseDisplayDate(v){
 const s=(v||'').trim();const match=s.match(/^(\d{1,2})[\/.-](\d{1,2})[\/.-](\d{2}|\d{4})$/);
 if(!match)return '';
 const day=Number(match[1]),month=Number(match[2]),year=match[3].length===2?2000+Number(match[3]):Number(match[3]);
 if(year!==activeYear||month<1||month>12||day<1||day>31)return '';
 const d=new Date(Date.UTC(year,month-1,day));if(d.getUTCFullYear()!==year||d.getUTCMonth()+1!==month||d.getUTCDate()!==day)return '';
 return `${year}-${String(month).padStart(2,'0')}-${String(day).padStart(2,'0')}`;
}

function warningCollection(){
 const periods=['monthly',...months.map((_,i)=>String(i+1))];
 const snapshots=periods.map(period=>capacityWarningEvents(period));
 const all=snapshots.flatMap(s=>s.events);
 const priority={over:0,pending:1,near:2};
 const order=(a,b)=>{
  const importance=priority[a.level]-priority[b.level];
  if(importance)return importance;
  if(a.level==='over'||a.level==='near'){
   const severity=(b.loadRatio||0)-(a.loadRatio||0);
   if(Math.abs(severity)>0.00001)return severity;
  }
  const number=p=>p==='monthly'?0:Number(p);
  return number(a.period)-number(b.period)||a.title.localeCompare(b.title,'id')||a.description.localeCompare(b.description,'id');
 };
 all.sort(order);
 const totals={over:0,near:0,pending:0};
 all.forEach(e=>totals[e.level]++);
 return {all,totals,hasData:snapshots.some(s=>s.hasData)};
}

 return {teamsMissing,krosMissing,assignedTeamCount:demoTeams.length-teamsMissing.length,assignedKroCount:POK_KROS.length-krosMissing.length,teamCount:demoTeams.length,kroCount:POK_KROS.length};
